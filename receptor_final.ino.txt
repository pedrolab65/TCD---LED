#define PINO_LED 25

void setup() {
  Serial.begin(115200);
  Serial2.begin(9600, SERIAL_8N1, 16, 17);

  ledcAttach(PINO_LED, 1000, 8);
}

void loop() {
  if (Serial2.available()) {

    String msg = Serial2.readStringUntil('\n');
    int leitura = msg.toInt();

    int pwm = leitura / 16;

    ledcWrite(PINO_LED, pwm);

    Serial.print("PWM: ");
    Serial.println(pwm);
  }
}
